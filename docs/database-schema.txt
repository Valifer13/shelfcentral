table libraries {
  id int pk
  name varchar
  slug varchar [unique]
  subscription_plan enum('FREE', 'PRO', 'ENTERPRISE')
  is_active boolean
  settings json
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp
}

table users {
  id int pk
  library_id int [null]
  name varchar
  email varchar
  password varchar
  role enum('SUPER_ADMIN', 'ADMIN', 'LIBRARIAN', 'MEMBER') [default: 'MEMBER']
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp

  // Index: Composite Unique (library_id, email)
}

table authors {
  id int pk
  library_id int
  name varchar
  created_at timestamp
  updated_at timestamp
}

table publishers {
  id int pk
  library_id int
  name varchar
  created_at timestamp
  updated_at timestamp
}

table categories {
  id int pk
  library_id int
  name varchar
}

table books {
  id int pk
  library_id int
  isbn varchar
  title varchar
  publisher_id int
  category_id int
  publication_year int
  stock_total int
  stock_available int
  average_rating decimal(3,2)
  total_reviews int
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp

  // Index: Composite Unique (library_id, isbn)
}

table book_authors {
  id int pk
  book_id int
  author_id int

  // Index: Composite Primary/Unique (book_id, author_id)
}

table book_copies {
  id int pk
  book_id int
  barcode varchar unique
  status enum('AVAILABLE', 'BORROWED', 'LOST', 'MAINTENANCE', 'RESERVED')
  location varchar
  created_at timestamp
  updated_at timestamp

  // Index: Composite Index (book_id, status)
}

table borrows {
  id int pk
  library_id int
  book_copy_id int
  user_id int
  borrowed_at timestamp
  due_at timestamp
  returned_at timestamp
  total_fine decimal(12,2)
  status enum('ONGOING', 'RETURNED', 'OVERDUE', 'LOST')
  created_at timestamp

  // Index: Composite Index (library_id, user_id)
}

table fines {
  id int pk
  borrow_id int
  amount decimal(12,2)
  reason varchar
  status enum('UNPAID', 'PAID', 'WAIVED')
  created_at timestamp
}

table fine_settings {
  id int pk
  library_id int
  grace_period_days int
  fine_per_day decimal(12,2)
  max_fine_amount decimal(12,2)
  created_at timestamp
  updated_at timestamp
}

table reviews {
  id int pk
  library_id int
  user_id int
  book_id int
  rating tinyint // Constraint: CHECK (rating >= 1 AND rating <= 5)
  comment text
  created_at timestamp

  // Index: Composite Unique (user_id, book_id)
}

table favorites {
  id int pk
  library_id int
  user_id int
  book_id int
  created_at timestamp
  updated_at timestamp
}

table reservations {
  id int pk
  library_id int
  user_id int
  book_id int
  book_copy_id int
  status enum('PENDING', 'READY_FOR_PICKUP', 'EXPIRED', 'COMPLETED', 'CANCELLED')
  queue_priority integer
  notified_at timestamp
  expires_at timestamp
  created_at timestamp
}

table audit_logs {
  id uuid pk
  library_id int
  user_id int
  action_type varchar
  source_ip varchar
  resource_type varchar // misal: 'BOOK', 'USER', 'BORROW'
  resource_id varchar   // Pakai varchar agar fleksibel untuk UUID atau INT
  status varchar
  created_at timestamp
}
